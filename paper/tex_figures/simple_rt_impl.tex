\begin{sequencediagram}

    %–– Lifelines ––––––––––––––––––––––––––––––––––––––––––––––
    \newthread{img}{:Image Gen}
    \newinst  {bvh}{:BVH}
    \newinst  {mat}{:Material}

    %–– Outer loop: pixels × samples ––––––––––––––––––––––––––
    \begin{sdblock}{draw}{for each pixel $\times$ sample}

        \begin{call}{img}{hit(ray)}{bvh}{HitInfo}
            \begin{callself}{bvh}{hit(child)}{…}{}
            \end{callself}
        \end{call}

        %–– Inner loop: recursion / scatter ––––––––––––––––––
        \begin{sdblock}{scatter}{while depth > 0}

            \begin{call}{img}{scatter(HitInfo)}{mat}{child rays}
            \end{call}

            % self-call shows further bounces, condensed
            \begin{callself}{bvh}{hit(child)}{…}{}
            \end{callself}

        \end{sdblock}

    \end{sdblock}

\end{sequencediagram}
