\begin{mathalgo}{Boxing usertype}
auto _box = [&](T& obj) -> Value* {
    return new_struct<T>(                       // allocate Julia struct
        array<Value*, N>{                       // N == sizeof...(FLens)
            invoke(                             // pack-expand over FLens…
                match{
                    [&]<concrete FT>(FT& v){ 
                        return box<FT>(v); 
                    },
                    [&]<abstract FT>(FT& v){ 
                        return dispatch_vptr(v); 
                    }
                },
                FLens::get(obj)                 // extract field
            )...                                // …for every FLens
        }
    );
}
\end{mathalgo}
