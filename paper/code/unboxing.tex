\begin{mathalgo}{Unboxing usertype}
auto _unbox = [&](Value* in) -> T {
    T out;                                           

    for_each_indexed<FLens...>(
        [&]<auto FL, auto I>() { 
            dispatch_tptr<FL>(
                out, 
                get_field<FT, I>(in));          
        }
    );

    return out;                                         
}
\end{mathalgo}
