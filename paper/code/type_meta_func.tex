template <
    ct_string Name, 
    typename T, typename M, 
    M T::* Member
>
struct Lens<Name, Member> {
    using type  = internal_t<M>;
    
    static M T::* member_ = Member;

    static auto label() { 
        return Name; 
    }
    static auto get(T& obj) -> M { 
        return obj.*member_; 
    }
    template <typename U>
    static auto set(T& obj, U&& value) -> void {
        obj.*member_ = value;
    }
};
